cmake_minimum_required(VERSION 3.16)
project(wuaizhibo LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native-deps")
set(MEDIA_PREFIX "${DEPS}/media/prefix")
add_library(waimedia SHARED media_bridge.cpp)
target_include_directories(waimedia PRIVATE "${DEPS}/headers" "${MEDIA_PREFIX}/include")
foreach(component avformat avcodec avutil swresample swscale)
  add_library(media_${component} SHARED IMPORTED)
  set_target_properties(media_${component} PROPERTIES IMPORTED_LOCATION "${MEDIA_PREFIX}/lib/lib${component}.so")
  target_link_libraries(waimedia PRIVATE media_${component})
endforeach()
target_link_libraries(waimedia PRIVATE libace_napi.z.so libhilog_ndk.z.so libnative_window.so libEGL.so libGLESv3.so libohaudio.so)
target_compile_options(waimedia PRIVATE -Wall -Wextra)
